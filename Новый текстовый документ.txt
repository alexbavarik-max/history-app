        let isSoundEnabled = true;

        function playSynthTone(frequency, duration, type = 'sine', volume = 0.3) {
            if (!isSoundEnabled) return;
            try {
                const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                const oscillator = audioCtx.createOscillator();
                const gainNode = audioCtx.createGain();
                oscillator.type = type;
                oscillator.frequency.value = frequency;
                gainNode.gain.setValueAtTime(volume, audioCtx.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
                oscillator.connect(gainNode);
                gainNode.connect(audioCtx.destination);
                oscillator.start();
                oscillator.stop(audioCtx.currentTime + duration);
            } catch (e) {}
        }

        function playClick() {
            if (!isSoundEnabled) return;
            playSynthTone(800, 0.08, 'sine', 0.35);
        }

        function playCorrectSound() {
            if (!isSoundEnabled) return;
            playSynthTone(523, 0.15, 'sine', 0.5);
            setTimeout(() => playSynthTone(659, 0.15, 'sine', 0.5), 150);
            setTimeout(() => playSynthTone(784, 0.25, 'sine', 0.6), 300);
        }

        function playWrongSound() {
            if (!isSoundEnabled) return;
            playSynthTone(400, 0.3, 'sawtooth', 0.35);
            setTimeout(() => playSynthTone(300, 0.4, 'sawtooth', 0.35), 250);
        }

        // ============================================================
        // ===== УПРАВЛЕНИЕ ЗВУКОМ (одна кнопка в меню) =====
        // ============================================================
        let bgMusic = null;
        let isBgMusicPlaying = false;
        let bgMusicInterval = null;
        let musicStarted = false;

        function initBgMusic() {
            try {
                bgMusic = new Audio('bg-music.mp3');
                bgMusic.loop = true;
                bgMusic.volume = 0.4;
                bgMusic.preload = 'auto';
                bgMusic.addEventListener('error', () => {
                    bgMusic = null;
                });
            } catch (e) {
                bgMusic = null;
            }
        }

        function toggleSoundAndMusic() {
            isSoundEnabled = !isSoundEnabled;
            
            if (!isSoundEnabled) {
                stopBgMusic();
                updateMusicButton('off');
            } else {
                updateMusicButton('on');
                if (!isBgMusicPlaying) {
                    startBgMusic();
                }
            }
        }

        function startBgMusic() {
            if (!isSoundEnabled) return;
            if (isBgMusicPlaying) return;
            
            isBgMusicPlaying = true;
            musicStarted = true;
            
            if (bgMusic) {
                try {
                    bgMusic.currentTime = 0;
                    bgMusic.play().catch(() => {
                        playSynthBgMusic();
                    });
                    return;
                } catch (e) {
                    playSynthBgMusic();
                }
            } else {
                playSynthBgMusic();
            }
        }

        function playSynthBgMusic() {
            const notes = [523, 587, 659, 784];
            let noteIndex = 0;
            
            if (bgMusicInterval) {
                clearInterval(bgMusicInterval);
                bgMusicInterval = null;
            }
            
            bgMusicInterval = setInterval(() => {
                if (!isBgMusicPlaying || !isSoundEnabled) return;
                try {
                    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                    const oscillator = audioCtx.createOscillator();
                    const gainNode = audioCtx.createGain();
                    oscillator.type = 'sine';
                    oscillator.frequency.value = notes[noteIndex % notes.length];
                    gainNode.gain.setValueAtTime(0.15, audioCtx.currentTime);
                    gainNode.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
                    oscillator.connect(gainNode);
                    gainNode.connect(audioCtx.destination);
                    oscillator.start();
                    oscillator.stop(audioCtx.currentTime + 0.4);
                    noteIndex++;
                } catch (e) {}
            }, 500);
        }

        function stopBgMusic() {
            isBgMusicPlaying = false;
            musicStarted = false;
            if (bgMusic) {
                try {
                    bgMusic.pause();
                    bgMusic.currentTime = 0;
                } catch (e) {}
            }
            if (bgMusicInterval) {
                clearInterval(bgMusicInterval);
                bgMusicInterval = null;
            }
        }

        function updateMusicButton(state) {
            const btn = document.getElementById('musicToggleBtn');
            if (state === 'on') {
                btn.innerHTML = '<span class="icon">🔊</span><span class="label">Звук: Вкл</span>';
            } else {
                btn.innerHTML = '<span class="icon">🔇</span><span class="label">Звук: Выкл</span>';
            }
        }

        // ===== ЗАПУСК МУЗЫКИ СРАЗУ ПОСЛЕ ЗАГРУЗКИ =====
        function autoStartMusic() {
            if (!musicStarted && isSoundEnabled) {
                startBgMusic();
                updateMusicButton('on');
            }
        }

        // Пробуем запустить музыку сразу
        initBgMusic();

        // Запускаем музыку с небольшой задержкой (чтобы страница загрузилась)
        setTimeout(() => {
            autoStartMusic();
        }, 300);

        // Если браузер заблокировал — запускаем после первого клика
        document.addEventListener('click', () => {
            if (!musicStarted && isSoundEnabled) {
                startBgMusic();
                updateMusicButton('on');
            }
        });

        function sliceDate(text) {
            let match = text.match(/\b([А-Яа-я]+\s+)?\d{4}\b/);
            if (match) {
                let idx = text.indexOf(match[0]);
                let endIdx = idx + match[0].length;
                let rest = text.substring(endIdx);
                let rangeMatch = rest.match(/^\s*[-–—]\s*\d{4}/);
                if (rangeMatch) {
                    endIdx += rangeMatch[0].length;
                }
                return text.substring(0, endIdx).trim();
            }
            return text;
        }

        function isPeriod(text) {
            return /[-–—]\s*\d{4}/.test(text);
        }

        function extractYear(text) {
            let match = text.match(/\b(\d{4})\b/);
            if (match) return parseInt(match[0]);
            return null;
        }

        function extractEvent(text) {
            let datePart = sliceDate(text);
            if (datePart) {
                let event = text.substring(datePart.length).trim();
                event = event.replace(/^[-–—\s]+/, '');
                if (event) return event;
            }
            return text;
        }

        function shuffleArray(arr) {
            for (let i = arr.length - 1; i > 0; i--) {
                let j = Math.floor(Math.random() * (i + 1));
                [arr[i], arr[j]] = [arr[j], arr[i]];
            }
            return arr;
        }

        function showQuizQuestion(type) {
            let available = dates.filter(d => !usedDatesForQuiz.includes(d));
            if (available.length < 4) {
                usedDatesForQuiz = [];
                available = dates.filter(d => !usedDatesForQuiz.includes(d));
                if (available.length < 4) {
                    document.getElementById('questionBox').innerHTML = '<div class="date">🎉 Все даты изучены!</div>';
                    document.getElementById('optionsContainer').innerHTML = '';
                    document.getElementById('resultContainer').innerHTML = '';
                    document.getElementById('nextQuizBtn').classList.add('hidden');
                    document.getElementById('quizHint').classList.remove('hidden');
                    return;
                }
            }

            let correctDate = available[Math.floor(Math.random() * available.length)];
            usedDatesForQuiz.push(correctDate);

            let datePart = sliceDate(correctDate);
            let correctEvent = extractEvent(correctDate);
            let correctYear = extractYear(correctDate);
            let correctIsPeriod = isPeriod(datePart);

            if (!correctEvent || correctYear === null) {
                showQuizQuestion(type);
                return;
            }

            let options = [];
            let otherDates = dates.filter(d => d !== correctDate && !usedDatesForQuiz.includes(d));
            shuffleArray(otherDates);

            if (type === 'date-to-event') {
                document.getElementById('questionBox').innerHTML = `<div class="date">📅 ${datePart}</div>`;
                options = [correctEvent];
                
                let nearby = [];
                for (let d of otherDates) {
                    let y = extractYear(d);
                    let e = extractEvent(d);
                    if (y !== null && e && Math.abs(y - correctYear) <= 5 && e !== correctEvent) {
                        nearby.push(e);
                    }
                }
                if (nearby.length >= 3) {
                    shuffleArray(nearby);
                    options.push(...nearby.slice(0, 3));
                } else {
                    for (let d of otherDates.slice(0, 3)) {
                        let e = extractEvent(d);
                        if (e && e !== correctEvent) options.push(e);
                    }
                }
                while (options.length < 4) {
                    options.push('Фиктивное событие ' + (correctYear + Math.floor(Math.random() * 10 - 5)));
                }
            } else {
                document.getElementById('questionBox').innerHTML = `<div class="event">📖 ${correctEvent}</div>`;
                options = [datePart];
                
                let nearby = [];
                for (let d of otherDates) {
                    let y = extractYear(d);
                    let dDate = sliceDate(d);
                    let isPeriodCandidate = isPeriod(dDate);
                    if (y !== null && Math.abs(y - correctYear) <= 5 && isPeriodCandidate === correctIsPeriod) {
                        nearby.push(dDate);
                    }
                }
                if (nearby.length >= 3) {
                    shuffleArray(nearby);
                    options.push(...nearby.slice(0, 3));
                } else {
                    for (let d of otherDates.slice(0, 3)) {
                        let s = sliceDate(d);
                        let isPeriodCandidate = isPeriod(s);
                        if (s !== datePart && isPeriodCandidate === correctIsPeriod) {
                            options.push(s);
                        }
                    }
                }
                while (options.length < 4) {
                    let fake;
                    if (correctIsPeriod) {
                        let start = correctYear + Math.floor(Math.random() * 6 - 3);
                        let end = start + Math.floor(Math.random() * 4 + 1);
                        fake = start + '-' + end;
                    } else {
                        fake = String(correctYear + Math.floor(Math.random() * 10 - 5));
                    }
                    if (fake !== datePart) options.push(fake);
                }
            }

            let uniqueOptions = [...new Set(options)];
            while (uniqueOptions.length < 4) {
                let extraFake;
                if (type === 'date-to-event') {
                    extraFake = 'Фиктивное событие ' + (correctYear + Math.floor(Math.random() * 10 - 5));
                } else {
                    if (correctIsPeriod) {
                        let start = correctYear + Math.floor(Math.random() * 6 - 3);
                        let end = start + Math.floor(Math.random() * 4 + 1);
                        extraFake = start + '-' + end;
                    } else {
                        extraFake = String(correctYear + Math.floor(Math.random() * 10 - 5));
                    }
                }
                if (!uniqueOptions.includes(extraFake) && extraFake !== datePart) {
                    uniqueOptions.push(extraFake);
                }
            }

            shuffleArray(uniqueOptions);

            let container = document.getElementById('optionsContainer');
            container.innerHTML = '';
            uniqueOptions.forEach((opt, i) => {
                let btn = document.createElement('button');
                btn.className = 'option-btn';
                btn.innerHTML = `<span class="num">${i+1}</span>${opt}`;
                btn.onclick = () => {
                    playClick();
                    checkAnswer(opt, correctDate, type, uniqueOptions);
                };
                container.appendChild(btn);
            });

            document.getElementById('resultContainer').innerHTML = '';
            document.getElementById('nextQuizBtn').classList.add('hidden');
            document.getElementById('quizHint').classList.remove('hidden');
            
            nextQuestionData = { type, correctDate };
        }

        function checkAnswer(selected, correctDate, type, options) {
            let datePart = sliceDate(correctDate);
            let correctEvent = extractEvent(correctDate);
            let isCorrect = false;

            if (type === 'date-to-event') {
                isCorrect = (selected === correctEvent);
            } else {
                isCorrect = (selected === datePart);
            }

            let container = document.getElementById('resultContainer');
            let btns = document.querySelectorAll('.option-btn');
            
            btns.forEach(btn => btn.classList.add('disabled'));
            
            btns.forEach((btn) => {
                let text = btn.textContent.substring(btn.textContent.indexOf('.') + 2);
                if (type === 'date-to-event') {
                    if (text === correctEvent) btn.classList.add('show-correct');
                } else {
                    if (text === datePart) btn.classList.add('show-correct');
                }
                if (btn.textContent.includes(selected)) {
                    if (!isCorrect) btn.classList.add('wrong');
                }
            });

            if (isCorrect) {
                correct++;
                playCorrectSound();
                container.innerHTML = `
                    <div class="result-box correct">
                        <div class="icon">✅</div>
                        <div class="text">ПРАВИЛЬНО!</div>
                        <div class="detail">${correctDate}</div>
                    </div>
                `;
            } else {
                incorrect++;
                playWrongSound();
                container.innerHTML = `
                    <div class="result-box wrong">
